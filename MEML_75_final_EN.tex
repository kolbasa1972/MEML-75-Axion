\begin{abstract}
\noindent
We present model MEML~75, which describes the Standard Model
fermion sector through the geometry of the six-dimensional
orbifold $T^2/\mathbb{Z}_3$ with two independent tori.
The model contains \textbf{13 free parameters} and reproduces
\textbf{15 observables}: 9 quark (5 masses $+$ 3 CKM $+$ $J$),
4 lepton (PMNS), and 2 cosmological ($\nu_a$, $\Omega_a h^2$).

\textbf{The main testable prediction:}
axion dark matter with
$f_a = 1.9801\times10^{11}$~GeV,
$m_a = 28.407$~$\mu$eV, $\nu_a = 6.8688$~GHz, and axion-photon
coupling
$g_{a\gamma\gamma} = +4.38\times10^{-15}$~GeV$^{-1}$ (DFSZ)
or $-1.13\times10^{-14}$~GeV$^{-1}$ (KSVZ), testable
at ADMX/HAYSTAC (0.5--10~GHz).

\textbf{Classification of quantities:}
$\alpha_G = 2/(3\pi)$ is a \emph{postulate} with geometric
motivation; $\varepsilon = (\alpha_G/(4\pi))^2$ is a one-loop
\emph{ansatz}; $m_{KK}$, $M_6$ are conditionally derived at
$\mathcal{I} = 0.3653$.

\textbf{Fermion sector.} A global fit with loss function
$\chi^2 = \sum_i ((O_i-O_i^{\rm target})/\sigma_i)^2$ yields
$\chi^2 = 6.195\times10^{-20}$ at
$\tau_u = -0.1577944085 + 2.3851782590\,i$,
$\tau_d = +0.3693905617 + 1.5410496057\,i$.
Independent reproduction is given in~\S\ref{sec:fermion}
and Appendix~C. $\mathrm{Im}\,\tau$ is stable ($\pm 0.3\%$),
$\mathrm{Re}\,\tau$ is soft-flat (Appendix~E).

\textbf{Systematic tests} (all negative):
(1)~the universal Yukawa ansatz is excluded;
(2)~36 matrix structures are equally unstable (spread of
$|V_{ub}|$ from $-97\%$ to $+25407\%$);
(3)~$\theta_i$ from inflation is excluded at $26\sigma$;
(4)~modular forms at $\tau = e^{i\pi/3}$ do not yield
$\alpha_G$ exactly ($|Y^{(6)}_2|^2 = 0.445302 \neq 4/9$).

\textbf{Coefficient degeneracy.} At $N_{\rm dof} = 9-11 = -2$
the coefficients $c_i$ are non-unique: $\chi^2$ is invariant
under $c_i \to k c_i$ (Appendix~D). The model
\emph{interpolates} observations, it does not predict.

\textbf{Verdict:} the model is \emph{phenomenological}.
\end{abstract}

Modular models of fermion masses~\cite{Feruglio2017,Kobayashi2019}
build the mass hierarchy and mixing from compactification geometry.

independent tori $T^2/\mathbb{Z}_3$ with moduli $\tau_u$ and
$\tau_d$, and the mass matrices are built from $A_4$ modular
forms of weights $(2,4,6)$. The axion arising from the same
geometric ingredients provides dark matter.

\subsection{Metric and modular parameter}

The orbifold $T^2/\mathbb{Z}_3$:
\begin{equation}
z = x + \tau y, \quad ds^2 = dx^2 + dx\,dy + dy^2,
\end{equation}
where $\tau = e^{i\pi/3} = \frac12 + i\frac{\sqrt3}{2}$.

\subsection{Fixed points}

$Z_3$-action $z \to \omega z$, $\omega = e^{2\pi i/3}$:
\begin{equation}
z_1 = (0,0), \quad z_2 = (\tfrac13,\tfrac13), \quad z_3 = (\tfrac23,\tfrac23).
\end{equation}
These three points provide a topological explanation for three
generations.

\subsection{Two-torus structure}
\label{sec:tau-fit}

Quarks and leptons are localized on different tori with moduli:
\begin{equation}
\boxed{\;\tau_u = -0.1577944085 + 2.3851782590\,i, \quad
\tau_d = +0.3693905617 + 1.5410496057\,i\;}
\label{eq:tau-fit}
\end{equation}
\textbf{Status: [FIT].} Determined by the global fit
(\S\ref{sec:fermion}, Appendix~B). The values are given to 10
significant digits because the minimum is razor-thin: rounding to
3 significant digits raises $\chi^2$ from $6\times10^{-20}$ to
$2.6\times10^{-3}$ (Appendix~C).
$\mathrm{Im}\,\tau_u = 2.385$, $\mathrm{Im}\,\tau_d = 1.541$ are
stable across all fits; $\mathrm{Re}\,\tau$ is soft-flat
(Appendix~E).

\subsection{Matching 6D$\leftrightarrow$4D}

Holomorphic Yukawa coupling:
$Y_{\rm hol}(\tau) = \sum_n N_n e^{-2\pi n\,{\rm Im}\,\tau} P_n(\tau)$.

\begin{equation}
Y_{\rm phys} \sim e^{-S_{\rm inst}}\,({\rm Im}\,\tau)^{\Delta k},
\quad \Delta k = \tfrac{k_i+k_j+k_k}{2} - b,

with $\Delta k = 1$ (up), $\Delta k = 2$ (down).

Numerical check:
$e^{-\pi^2} = 5.1723\times10^{-5}$;
$\varepsilon/e^{-\pi^2} = 5.514$;
$(2\,{\rm Im}\,\tau_u)^1 = 4.774$;
$(2\,{\rm Im}\,\tau_d)^2 = 6.248$.
$5.514 \in [4.774, 6.248]$ --- within range.

\section{Conclusions}
% ============================================================

\textbf{What has been done:}
\begin{enumerate}[leftmargin=1.5em,itemsep=0.05em]
\item A model with 13 parameters for 15 observables.
\item 4 systematic tests (B.1, C.3, B.2, A.1) --- all negative.
\item A global fit $\chi^2 = 6.195\times10^{-20}$ at
      $\tau_u = -0.1578+2.3852i$, $\tau_d = +0.3694+1.5410i$;
      independently reproduced (3 DE runs).
\item A test of coefficient degeneracy: $\chi^2$ is invariant
      under $c_i \to kc_i$.
\item $\Omega_a h^2$: bare $\to$ anh $\to$ num, $\delta_T = +12.4\%$.
\item $g_{a\gamma\gamma}$ (KSVZ/DFSZ) --- two-point target for ADMX.
\item AIC/BIC with $N_{\rm dof} < 0$ caveat.
\end{enumerate}

\textbf{What works:} PMNS ($<0.5\%$), CKM ($\chi^2 \sim 10^{-20}$, fit),
Planck compatibility of dark matter, testable prediction
$\nu_a = 6.8688$~GHz,
$g_{a\gamma\gamma} \in \{+4.38\times10^{-15}, -1.13\times10^{-14}\}$~GeV$^{-1}$.

\textbf{What does not work:} LOO is unstable; coefficients are
scale-degenerate; $\alpha_G$, $\varepsilon$, $\theta_i$ are not
derived from first principles.

\textbf{Verdict.} The model is \emph{phenomenological}. It
reproduces observations but does not predict. The coefficient
degeneracy shows that the model interpolates 9 observations with
11 parameters.

\textbf{Priority for future work:} a mechanism to fix $c_i$.
Flux is only needed for $\mathrm{Im}\,\tau$ at $\pm 3\%$.

\section{Systematic search for $\alpha_G$}
\label{app:alpha-search}

Basis: 16 elements ($Y^{(2)}_i$, $Y^{(4)}_i$, $Y^{(6)}_i$, $\eta$, $\pi$, $\sqrt\pi$, $\pi^2$, $e^{-\pi^2}$),
where $\eta(\tau) = q^{1/24}\prod_{n=1}^\infty (1-q^n)$ is the
Dedekind eta function.
Tested $N = 392$ combinations:
$|a|/|b|$ (240), $|a|\cdot|b|$ (120), $\sqrt{|a|/\pi}$ (16), $|a|^2/\pi^{3/2}$ (16).

\textbf{2 coincidences below $0.1\%$:}
$|Y^{(6)}_1|/|Y^{(4)}_3| = 0.212125$ ($-0.0385\%$, $n=5..30$);
$|Y^{(6)}_2|/\pi = 0.212411$ ($+0.0965\%$, $n=5..30$).

\textbf{Poisson:} $\langle n\rangle = 392 \cdot 0.002 = 0.78$;
$n_{\rm obs} = 2$; $P(n \ge 2) = 0.185$ (not significant).

\textbf{Conclusion:} $\alpha_G$ remains a postulate.
